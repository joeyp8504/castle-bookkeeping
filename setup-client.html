New Client Setup

Create a Google Drive folder and client portal in one flow.

1

Deploy the Drive Folder Script (One Time Only)

Copy the code below and deploy it as a Google Apps Script web app. You only need to do this once.

  1. Go to script.google.com (sign in with your Castle Google account)
  2. Click New Project
  3. Delete the default code and paste the code below
  4. Click Deploy → New Deployment
  5. Type: Web app
  6. Execute as: Me
  7. Who has access: Anyone
  8. Click Deploy and authorize when prompted
  9. Copy the Web app URL and paste it in Step 2 below
function doGet(e) { return handleRequest(e); } function doPost(e) { return handleRequest(e); } function handleRequest(e) { var params = e.parameter; var clientName = params.name || 'New Client'; // Find or create the parent folder var parentName = 'Castle Clients'; var parentFolders = DriveApp.getFoldersByName(parentName); var parent; if (parentFolders.hasNext()) { parent = parentFolders.next(); } else { parent = DriveApp.createFolder(parentName); } // Create client folder var clientFolder = parent.createFolder(clientName); // Create subfolders clientFolder.createFolder('Bank Statements'); clientFolder.createFolder('Revenue Records'); clientFolder.createFolder('Expense Receipts'); clientFolder.createFolder('Payroll Records'); clientFolder.createFolder('Other Documents'); // Set sharing: anyone with link can edit clientFolder.setSharing( DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.EDIT ); var folderUrl = clientFolder.getUrl(); var folderId = clientFolder.getId(); // Return JSON var output = { success: true, clientName: clientName, folderUrl: folderUrl, folderId: folderId, subfolders: [ 'Bank Statements', 'Revenue Records', 'Expense Receipts', 'Payroll Records', 'Other Documents' ] }; return ContentService .createTextOutput(JSON.stringify(output)) .setMimeType(ContentService.MimeType.JSON); }
